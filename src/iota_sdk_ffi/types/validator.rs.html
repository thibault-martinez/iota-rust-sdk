15/// The Validator Set for a particular epoch.
16///
17/// # BCS
18///
19/// The BCS serialized form for this type is defined by the following ABNF:
20///
21/// ```text
22/// validator-committee = u64 ; epoch
23///                       (vector validator-committee-member)
24/// ```
25#[derive(uniffi::Record)]
26pub struct ValidatorCommittee {
27    pub epoch: EpochId,
28    pub members: Vec<ValidatorCommitteeMember>,
29}
30
31impl From<iota_sdk::types::ValidatorCommittee> for ValidatorCommittee {
32    fn from(value: iota_sdk::types::ValidatorCommittee) -> Self {
33        Self {
34            epoch: value.epoch,
35            members: value.members.into_iter().map(Into::into).collect(),
36        }
37    }
38}
39
40impl From<ValidatorCommittee> for iota_sdk::types::ValidatorCommittee {
41    fn from(value: ValidatorCommittee) -> Self {
42        Self {
43            epoch: value.epoch,
44            members: value.members.into_iter().map(Into::into).collect(),
45        }
46    }
47}

49/// A member of a Validator Committee
50///
51/// # BCS
52///
53/// The BCS serialized form for this type is defined by the following ABNF:
54///
55/// ```text
56/// validator-committee-member = bls-public-key
57///                              u64 ; stake
58/// ```
59#[derive(Clone, uniffi::Record)]
60pub struct ValidatorCommitteeMember {
61    pub public_key: Arc<Bls12381PublicKey>,
62    pub stake: u64,
63}
64
65impl From<iota_sdk::types::ValidatorCommitteeMember> for ValidatorCommitteeMember {
66    fn from(value: iota_sdk::types::ValidatorCommitteeMember) -> Self {
67        Self {
68            public_key: Arc::new(value.public_key.into()),
69            stake: value.stake,
70        }
71    }
72}
73
74impl From<ValidatorCommitteeMember> for iota_sdk::types::ValidatorCommitteeMember {
75    fn from(value: ValidatorCommitteeMember) -> Self {
76        Self {
77            public_key: **value.public_key,
78            stake: value.stake,
79        }
80    }
81}

83/// A signature from a Validator
84///
85/// # BCS
86///
87/// The BCS serialized form for this type is defined by the following ABNF:
88///
89/// ```text
90/// validator-signature = u64               ; epoch
91///                       bls-public-key
92///                       bls-signature
93/// ```
94#[derive(derive_more::From, uniffi::Object)]
95pub struct ValidatorSignature(pub iota_sdk::types::ValidatorSignature);
96
97#[uniffi::export]
98impl ValidatorSignature {
99    #[uniffi::constructor]
100    pub fn new(
101        epoch: EpochId,
102        public_key: &Bls12381PublicKey,
103        signature: &Bls12381Signature,
104    ) -> Self {
105        Self(iota_sdk::types::ValidatorSignature {
106            epoch,
107            public_key: **public_key,
108            signature: **signature,
109        })
110    }
111
112    pub fn epoch(&self) -> EpochId {
113        self.0.epoch
114    }
115
116    pub fn public_key(&self) -> Bls12381PublicKey {
117        self.0.public_key.into()
118    }
119
120    pub fn signature(&self) -> Bls12381Signature {
121        self.0.signature.into()
122    }
123}

125/// An aggregated signature from multiple Validators.
126///
127/// # BCS
128///
129/// The BCS serialized form for this type is defined by the following ABNF:
130///
131/// ```text
132/// validator-aggregated-signature = u64               ; epoch
133///                                  bls-signature
134///                                  roaring-bitmap
135/// roaring-bitmap = bytes  ; where the contents of the bytes are valid
136///                         ; according to the serialized spec for
137///                         ; roaring bitmaps
138/// ```
139///
140/// See <https://github.com/RoaringBitmap/RoaringFormatSpec> for the specification for the
141/// serialized format of RoaringBitmaps.
142#[derive(derive_more::From, uniffi::Object)]
143pub struct ValidatorAggregatedSignature(pub iota_sdk::types::ValidatorAggregatedSignature);
144
145#[uniffi::export]
146impl ValidatorAggregatedSignature {
147    #[uniffi::constructor]
148    pub fn new(epoch: EpochId, signature: &Bls12381Signature, bitmap_bytes: &[u8]) -> Result<Self> {
149        Ok(Self(iota_sdk::types::ValidatorAggregatedSignature {
150            epoch,
151            signature: **signature,
152            bitmap: roaring::RoaringBitmap::deserialize_from(bitmap_bytes)?,
153        }))
154    }
155
156    pub fn epoch(&self) -> EpochId {
157        self.0.epoch
158    }
159
160    pub fn signature(&self) -> Bls12381Signature {
161        self.0.signature.into()
162    }
163
164    pub fn bitmap_bytes(&self) -> Result<Vec<u8>> {
165        let mut bytes = Vec::new();
166        self.0.bitmap.serialize_into(&mut bytes)?;
167        Ok(bytes)
168    }
169}
