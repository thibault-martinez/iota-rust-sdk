13/// The status of an executed Transaction
14///
15/// # BCS
16///
17/// The BCS serialized form for this type is defined by the following ABNF:
18///
19/// ```text
20/// execution-status = success / failure
21/// success = %x00
22/// failure = %x01 execution-error (option u64)
23/// ```
24#[derive(uniffi::Enum)]
25pub enum ExecutionStatus {
26    /// The Transaction successfully executed.
27    Success,
28    /// The Transaction didn't execute successfully.
29    ///
30    /// Failed transactions are still committed to the blockchain but any
31    /// intended effects are rolled back to prior to this transaction
32    /// executing with the caveat that gas objects are still smashed and gas
33    /// usage is still charged.
34    Failure {
35        /// The error encountered during execution.
36        error: ExecutionError,
37        /// The command, if any, during which the error occurred.
38        command: Option<u64>,
39    },
40}
41
42impl From<iota_sdk::types::ExecutionStatus> for ExecutionStatus {
43    fn from(value: iota_sdk::types::ExecutionStatus) -> Self {
44        match value {
45            iota_sdk::types::ExecutionStatus::Success => Self::Success,
46            iota_sdk::types::ExecutionStatus::Failure { error, command } => Self::Failure {
47                error: error.into(),
48                command,
49            },
50        }
51    }
52}

57            ExecutionStatus::Success => Self::Success,
58            ExecutionStatus::Failure { error, command } => Self::Failure {
59                error: error.into(),
60                command,
61            },
62        }
63    }

533/// Location in move bytecode where an error occurred

535/// # BCS
536///
537/// The BCS serialized form for this type is defined by the following ABNF:
538///
539/// ```text
540/// move-location = object-id identifier u16 u16 (option identifier)
541/// ```
542#[derive(uniffi::Record)]
543pub struct MoveLocation {
544    /// The package id
545    pub package: Arc<ObjectId>,
546    /// The module name
547    pub module: String,
548    /// The function index
549    pub function: u16,
550    /// Index into the code stream for a jump. The offset is relative to the
551    /// beginning of the instruction stream.
552    pub instruction: u16,
553    /// The name of the function if available
554    #[uniffi(default = None)]
555    pub function_name: Option<String>,
556}

563            function: value.function,
564            instruction: value.instruction,
565            function_name: value.function_name.map(|v| v.to_string()),
566        }
567    }
568}

582/// An error with an argument to a command
583///
584/// # BCS
585///
586/// The BCS serialized form for this type is defined by the following ABNF:
587///
588/// ```text
589/// command-argument-error =  type-mismatch
590///                        =/ invalid-bcs-bytes
591///                        =/ invalid-usage-of-pure-argument
592///                        =/ invalid-argument-to-private-entry-function
593///                        =/ index-out-of-bounds
594///                        =/ secondary-index-out-of-bound
595///                        =/ invalid-result-arity
596///                        =/ invalid-gas-coin-usage
597///                        =/ invalid-value-usage
598///                        =/ invalid-object-by-value
599///                        =/ invalid-object-by-mut-ref
600///                        =/ shared-object-operation-not-allowed
601///
602/// type-mismatch                               = %x00
603/// invalid-bcs-bytes                           = %x01
604/// invalid-usage-of-pure-argument              = %x02
605/// invalid-argument-to-private-entry-function  = %x03
606/// index-out-of-bounds                         = %x04 u16
607/// secondary-index-out-of-bound                = %x05 u16 u16
608/// invalid-result-arity                        = %x06 u16
609/// invalid-gas-coin-usage                      = %x07
610/// invalid-value-usage                         = %x08
611/// invalid-object-by-value                     = %x09
612/// invalid-object-by-mut-ref                   = %x0a
613/// shared-object-operation-not-allowed         = %x0b
614/// ```
615#[uniffi::remote(Enum)]
616pub enum CommandArgumentError {
617    /// The type of the value does not match the expected type
618    TypeMismatch,
619    /// The argument cannot be deserialized into a value of the specified type
620    InvalidBcsBytes,
621    /// The argument cannot be instantiated from raw bytes
622    InvalidUsageOfPureArgument,
623    /// Invalid argument to private entry function.
624    /// Private entry functions cannot take arguments from other Move functions.
625    InvalidArgumentToPrivateEntryFunction,
626    /// Out of bounds access to input or results
627    IndexOutOfBounds { index: u16 },
628    /// Out of bounds access to subresult
629    SecondaryIndexOutOfBounds { result: u16, subresult: u16 },
630    /// Invalid usage of result.
631    /// Expected a single result but found either no return value or multiple.
632    InvalidResultArity { result: u16 },
633    /// Invalid usage of Gas coin.
634    /// The Gas coin can only be used by-value with a TransferObjects command.
635    InvalidGasCoinUsage,
636    /// Invalid usage of move value.
637    //     Mutably borrowed values require unique usage.
638    //     Immutably borrowed values cannot be taken or borrowed mutably.
639    //     Taken values cannot be used again.
640    InvalidValueUsage,
641    /// Immutable objects cannot be passed by-value.
642    InvalidObjectByValue,
643    /// Immutable objects cannot be passed by mutable reference, &mut.
644    InvalidObjectByMutRef,
645    /// Shared object operations such a wrapping, freezing, or converting to
646    /// owned are not allowed.
647    SharedObjectOperationNotAllowed,
648    /// Invalid argument arity. Expected a single argument but found a result
649    /// that expanded to multiple arguments.
650    InvalidArgumentArity,
651}

661///                         not-a-package           /
662///                         incompatible-upgrade    /
663///                         digest-does-not-match   /
664///                         unknown-upgrade-policy  /
665///                         package-id-does-not-match
666///
667/// unable-to-fetch-package     = %x00 object-id
668/// not-a-package               = %x01 object-id
669/// incompatible-upgrade        = %x02
670/// digest-does-not-match       = %x03 digest
671/// unknown-upgrade-policy      = %x04 u8
672/// package-id-does-not-match   = %x05 object-id object-id
673/// ```
674#[derive(uniffi::Enum)]
675pub enum PackageUpgradeError {
676    /// Unable to fetch package
677    UnableToFetchPackage { package_id: Arc<ObjectId> },
678    /// Object is not a package
679    NotAPackage { object_id: Arc<ObjectId> },
680    /// Package upgrade is incompatible with previous version
681    IncompatibleUpgrade,
682    /// Digest in upgrade ticket and computed digest differ
683    DigestDoesNotMatch { digest: Arc<Digest> },
684    /// Upgrade policy is not valid
685    UnknownUpgradePolicy { policy: u8 },
686    /// PackageId does not matach PackageId in upgrade ticket
687    PackageIdDoesNotMatch {
688        package_id: Arc<ObjectId>,
689        ticket_id: Arc<ObjectId>,
690    },
691}
