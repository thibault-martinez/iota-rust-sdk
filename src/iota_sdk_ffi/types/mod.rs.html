24#[derive(derive_more::From, uniffi::Object)]
25pub struct PersonalMessage(pub(crate) iota_sdk::types::PersonalMessage<'static>);
26
27#[uniffi::export]
28impl PersonalMessage {
29    #[uniffi::constructor]
30    pub fn new(message_bytes: &[u8]) -> Self {
31        Self(iota_sdk::types::PersonalMessage(Cow::Owned(
32            message_bytes.to_vec(),
33        )))
34    }
35
36    pub fn message_bytes(&self) -> Vec<u8> {
37        self.0.0.clone().into_owned()
38    }
39
40    pub fn signing_digest(&self) -> Vec<u8> {
41        self.0.signing_digest().to_vec()
42    }
43
44    pub fn signing_digest_hex(&self) -> String {
45        self.0.signing_digest_hex()
46    }
47}
