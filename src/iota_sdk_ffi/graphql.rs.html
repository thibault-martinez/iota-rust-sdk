42#[uniffi::remote(Enum)]
43/// Determines what to wait for after executing a transaction.
44pub enum WaitForTx {
45    /// Indicates that the transaction effects will be usable in subsequent
46    /// transactions, and that the transaction itself is indexed on the node.
47    Indexed,
48    /// Indicates that the transaction has been included in a checkpoint, and
49    /// all queries may include it.
50    Finalized,
51}
