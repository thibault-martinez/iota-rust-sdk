28/// A builder for creating transactions. Use `finish` to finalize the
29/// transaction data.
30#[derive(derive_more::From, uniffi::Object)]
31pub struct TransactionBuilder(RwLock<iota_sdk::transaction_builder::TransactionBuilder<()>>);
32
33impl TransactionBuilder {
34    fn read<F, T>(&self, f: F) -> T
35    where
36        F: FnOnce(&iota_sdk::transaction_builder::TransactionBuilder<()>) -> T,
37    {
38        let lock = self.0.read().expect("error reading from builder");
39        f(&lock)
40    }
41
42    fn write<F, T>(&self, f: F) -> T
43    where
44        F: FnOnce(&mut iota_sdk::transaction_builder::TransactionBuilder<()>) -> T,
45    {
46        let mut lock = self.0.write().expect("error writing to builder");
47        f(&mut lock)
48    }
49}

98    /// Set the gas station sponsor.
99    #[uniffi::method(default(duration = None, headers = None))]
100    pub fn gas_station_sponsor(
101        self: Arc<Self>,
102        url: String,
103        duration: Option<Duration>,
104        headers: Option<HashMap<String, Vec<String>>>,
105    ) -> Arc<Self> {
106        self.write(|builder| {
107            let b = builder.gas_station_sponsor(url.parse().expect("invalid URL"));
108            if let Some(duration) = duration {
109                b.gas_reservation_duration(duration);
110            }
111            if let Some(headers) = headers {
112                for (name, values) in headers {
113                    for value in values {
114                        b.add_gas_station_header(
115                            name.parse().expect("invalid header name"),
116                            value.parse().expect("invalid header value"),
117                        );
118                    }
119                }
120            }
121        });
122        self
123    }

135    /// Call a Move function with the given arguments.
136    #[uniffi::method(default(type_args = [], arguments = [], names = []))]
137    pub fn move_call(
138        self: Arc<Self>,
139        package: &Address,
140        module: &Identifier,
141        function: &Identifier,
142        arguments: Vec<Arc<PTBArgument>>,
143        type_args: Vec<Arc<TypeTag>>,
144        names: Vec<String>,
145    ) -> Arc<Self> {
146        self.write(|builder| {
147            builder
148                .move_call(**package, &module.as_str(), &function.as_str())
149                .arguments(arguments)
150                .type_tags(type_args.into_iter().map(|v| v.0.clone()))
151                .name(names);
152        });
153        self
154    }

156    /// Send IOTA to a recipient address.

158    /// The `amount` parameter specifies the quantity in NANOS, where 1 IOTA

162        self.write(|builder| {
163            builder.send_iota(**recipient, amount);
164        });
165        self
166    }

168    /// Transfer some coins to a recipient address. If multiple coins are
169    /// provided then they will be merged.
170    ///
171    /// The `amount` parameter specifies the quantity in NANOS, where 1 IOTA
172    /// equals 1_000_000_000 NANOS.
173    /// If `amount` is provided, that amount is split from the provided coins
174    /// and sent.
175    /// If `amount` is `None`, the entire coins are transferred.
176    ///
177    /// All provided coins must have the same coin type. Mixing coins of
178    /// different types will result in an error.
179    ///
180    /// If you intend to transfer all provided coins to another address in a
181    /// single transaction, consider using
182    /// `TransactionBuilder::transfer_objects()` instead.
183    #[uniffi::method(default(amount = None))]
184    pub fn send_coins(
185        self: Arc<Self>,
186        coins: Vec<Arc<PTBArgument>>,
187        recipient: &Address,
188        amount: Option<Arc<PTBArgument>>,
189    ) -> Arc<Self> {
190        self.write(|builder| {
191            builder.send_coins::<_, &PTBArgument>(coins, **recipient, amount.as_deref());
192        });
193        self
194    }

225    /// This method combines the balances of multiple coins of the same coin

240    /// Make a move vector from a list of elements. The elements must all be of
241    /// the type indicated by `type_tag`.
242    pub fn make_move_vec(
243        self: Arc<Self>,
244        elements: Vec<Arc<MoveArg>>,
245        type_tag: &TypeTag,
246        name: String,
247    ) -> Arc<Self> {
248        use iota_sdk::transaction_builder::unresolved::{Command, MakeMoveVector};
249        self.write(|builder| {
250            let cmd = Command::MakeMoveVector(MakeMoveVector {
251                type_: Some(type_tag.0.clone()),
252                elements: elements
253                    .iter()
254                    .map(|e| builder.apply_argument(e.as_ref()))
255                    .collect(),
256            });
257            builder.named_command(cmd, name);
258        });
259        self
260    }

262    /// Publish a list of modules with the given dependencies. The result
263    /// assigned to `upgrade_cap_name` is the `0x2::package::UpgradeCap`
264    /// Move type. Note that the upgrade capability needs to be handled
265    /// after this call:
266    ///  - transfer it to the transaction sender or another address
267    ///  - burn it
268    ///  - wrap it for access control
269    ///  - discard the it to make a package immutable
270    ///
271    /// The arguments required for this command are:
272    ///  - `modules`: is the modules' bytecode to be published
273    ///  - `dependencies`: is the list of IDs of the transitive dependencies of
274    ///    the package
275    pub fn publish(
276        self: Arc<Self>,
277        package_data: &MovePackageData,
278        upgrade_cap_name: String,
279    ) -> Arc<Self> {
280        self.write(|builder| {
281            builder
282                .publish(package_data.0.clone())
283                .upgrade_cap(upgrade_cap_name);
284        });
285        self
286    }
287
288    /// Upgrade a Move package.
289    ///
290    ///  - `modules`: is the modules' bytecode for the modules to be published
291    ///  - `dependencies`: is the list of IDs of the transitive dependencies of
292    ///    the package to be upgraded
293    ///  - `package`: is the ID of the current package being upgraded
294    ///  - `ticket`: is the upgrade ticket
295    ///
296    ///  To get the ticket, you have to call the
297    /// `0x2::package::authorize_upgrade` function, and pass the package
298    /// ID, the upgrade policy, and package digest.
299    #[uniffi::method(default(name = None))]
300    pub fn upgrade(
301        self: Arc<Self>,
302        package_id: &ObjectId,
303        package_data: &MovePackageData,
304        upgrade_ticket: &PTBArgument,
305        name: Option<String>,
306    ) -> Arc<Self> {
307        self.write(|builder| {
308            builder
309                .upgrade(**package_id, package_data.0.clone(), upgrade_ticket)
310                .name(name);
311        });
312        self
313    }

334    /// Convert this builder into a transaction.
335    pub fn finish(&self) -> Result<Transaction> {
336        Ok(Transaction(self.read(|builder| builder.clone().finish())?))
337    }

339    /// Execute the transaction using the gas station and return the JSON
340    /// transaction effects. This will fail unless data is set with the
341    /// `gas_station_sponsor` function.
342    ///
343    /// NOTE: These effects are not necessarily compatible with
344    /// `TransactionEffects`
345    pub async fn execute_with_gas_station(
346        &self,
347        keypair: &SimpleKeypair,
348    ) -> Result<serde_json::Value> {
349        Ok(self
350            .read(|builder| builder.clone().execute_with_gas_station(&keypair.0))
351            .await?)
